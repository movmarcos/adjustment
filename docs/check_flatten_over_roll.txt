-- Check what a Flatten left behind on a COB that already carried a Roll (VaR).
-- Replace <COBID> (the target COB, e.g. 20260902) and <ENTITY>.
-- Add your book / department condition if the filter was narrower than the entity.
--
-- How to read the result (one row per adjustment still present at the COB):
--   * Roll rows remain and ROWS_WITH_NO_TARGET_ORIGINAL is most of them
--       -> positions the source COB had and the target did not; the Flatten
--          never touches them, so a residue stays in the combined view.
--   * Roll rows remain and ROWS_WITH_NO_TARGET_ORIGINAL is ~0
--       -> the supersede did not match those positions (engine bug to trace).
--   * The Flatten has no row at all
--       -> it failed or did not run; check its status in the grid / Logs page.

SELECT a.ADJUSTMENT_ID,
       d.ADJUSTMENT_TYPE,
       d.RUN_STATUS,
       COUNT(*)                        AS ROWS_,
       SUM(a.PNL_VECTOR_VALUE_IN_USD)  AS VALUE_USD,
       COUNT_IF(f.COBID IS NULL)       AS ROWS_WITH_NO_TARGET_ORIGINAL
FROM FACT.VAR_MEASURES_ADJUSTMENT a
LEFT JOIN DIMENSION.ADJUSTMENT d
  ON d.ADJUSTMENT_ID = a.ADJUSTMENT_ID
LEFT JOIN FACT.VAR_MEASURES f
  ON  f.COBID = a.COBID
  AND EQUAL_NULL(f.ENTITY_CODE, a.ENTITY_CODE)
  AND EQUAL_NULL(f.BOOK_KEY, a.BOOK_KEY)
  AND EQUAL_NULL(f.TRADE_KEY, a.TRADE_KEY)
  AND EQUAL_NULL(f.COMMON_INSTRUMENT_KEY, a.COMMON_INSTRUMENT_KEY)
  AND EQUAL_NULL(f.COMMON_INSTRUMENT_FCD_KEY, a.COMMON_INSTRUMENT_FCD_KEY)
  AND EQUAL_NULL(f.CURRENCY_CODE, a.CURRENCY_CODE)
  AND EQUAL_NULL(f.SOURCE_SYSTEM_CODE, a.SOURCE_SYSTEM_CODE)
  AND EQUAL_NULL(f.VAR_SUBCOMPONENT_ID, a.VAR_SUBCOMPONENT_ID)
  AND EQUAL_NULL(f.SCENARIO_DATE_ID, a.SCENARIO_DATE_ID)
  AND EQUAL_NULL(f.IS_OFFICIAL_SOURCE, a.IS_OFFICIAL_SOURCE)
WHERE a.COBID = <COBID>
  AND a.ENTITY_CODE = '<ENTITY>'
GROUP BY ALL
ORDER BY a.ADJUSTMENT_ID;

-- STEP 2 — which key column stops the Roll rows from matching the target COB?
-- For each surrogate-key column: how many of the Roll's rows have a value
-- that exists anywhere in the target COB's original rows for the entity.
-- A column at (or near) 0 is the one that differs per COB.
-- Replace <COBID> (target), <ENTITY>, <ROLL_ADJ_ID> (1015491 in the last run).
-- ═══════════════════════════════════════════════════════════════════════════
WITH t AS (
    SELECT * FROM FACT.VAR_MEASURES
    WHERE COBID = <COBID> AND ENTITY_CODE = '<ENTITY>'
),
r AS (
    SELECT * FROM FACT.VAR_MEASURES_ADJUSTMENT
    WHERE COBID = <COBID> AND ENTITY_CODE = '<ENTITY>' AND ADJUSTMENT_ID = <ROLL_ADJ_ID>
)
SELECT COUNT(*)                                                                    AS ROLL_ROWS,
       COUNT_IF(r.BOOK_KEY                  IN (SELECT BOOK_KEY                  FROM t)) AS BOOK_KEY_MATCH,
       COUNT_IF(r.TRADE_KEY                 IN (SELECT TRADE_KEY                 FROM t)) AS TRADE_KEY_MATCH,
       COUNT_IF(r.COMMON_INSTRUMENT_KEY     IN (SELECT COMMON_INSTRUMENT_KEY     FROM t)) AS COMMON_INSTRUMENT_KEY_MATCH,
       COUNT_IF(r.COMMON_INSTRUMENT_FCD_KEY IN (SELECT COMMON_INSTRUMENT_FCD_KEY FROM t)) AS COMMON_INSTRUMENT_FCD_KEY_MATCH,
       COUNT_IF(r.CURRENCY_CODE             IN (SELECT CURRENCY_CODE             FROM t)) AS CURRENCY_CODE_MATCH,
       COUNT_IF(r.SOURCE_SYSTEM_CODE        IN (SELECT SOURCE_SYSTEM_CODE        FROM t)) AS SOURCE_SYSTEM_CODE_MATCH,
       COUNT_IF(r.VAR_SUBCOMPONENT_ID       IN (SELECT VAR_SUBCOMPONENT_ID       FROM t)) AS VAR_SUBCOMPONENT_ID_MATCH,
       COUNT_IF(r.SCENARIO_DATE_ID          IN (SELECT SCENARIO_DATE_ID          FROM t)) AS SCENARIO_DATE_ID_MATCH,
       COUNT_IF(r.IS_OFFICIAL_SOURCE        IN (SELECT IS_OFFICIAL_SOURCE        FROM t)) AS IS_OFFICIAL_SOURCE_MATCH,
       COUNT_IF(r.TRADE_KEY IS NULL OR r.TRADE_KEY = -1)                            AS TRADE_KEY_NULL_OR_MINUS1,
       COUNT_IF(r.COMMON_INSTRUMENT_KEY IS NULL OR r.COMMON_INSTRUMENT_KEY = -1)    AS CI_KEY_NULL_OR_MINUS1
FROM r;


-- STEP 3 — do the dimension keys change between the two COBs? (SCD2 versions)
-- Replace <SOURCE_COBID> and <COBID>.
SELECT 'TRADE' AS DIM,
       COUNT(DISTINCT CASE WHEN TO_DATE('<SOURCE_COBID>','YYYYMMDD') BETWEEN EFFECTIVE_START_DATE AND EFFECTIVE_END_DATE THEN TRADE_KEY END) AS KEYS_AT_SOURCE,
       COUNT(DISTINCT CASE WHEN TO_DATE('<COBID>','YYYYMMDD')        BETWEEN EFFECTIVE_START_DATE AND EFFECTIVE_END_DATE THEN TRADE_KEY END) AS KEYS_AT_TARGET,
       COUNT(DISTINCT CASE WHEN TO_DATE('<SOURCE_COBID>','YYYYMMDD') BETWEEN EFFECTIVE_START_DATE AND EFFECTIVE_END_DATE
                            AND TO_DATE('<COBID>','YYYYMMDD')        BETWEEN EFFECTIVE_START_DATE AND EFFECTIVE_END_DATE THEN TRADE_KEY END) AS KEYS_VALID_AT_BOTH
FROM DIMENSION.TRADE
UNION ALL
SELECT 'COMMON_INSTRUMENT',
       COUNT(DISTINCT CASE WHEN TO_DATE('<SOURCE_COBID>','YYYYMMDD') BETWEEN EFFECTIVE_START_DATE AND EFFECTIVE_END_DATE THEN COMMON_INSTRUMENT_KEY END),
       COUNT(DISTINCT CASE WHEN TO_DATE('<COBID>','YYYYMMDD')        BETWEEN EFFECTIVE_START_DATE AND EFFECTIVE_END_DATE THEN COMMON_INSTRUMENT_KEY END),
       COUNT(DISTINCT CASE WHEN TO_DATE('<SOURCE_COBID>','YYYYMMDD') BETWEEN EFFECTIVE_START_DATE AND EFFECTIVE_END_DATE
                            AND TO_DATE('<COBID>','YYYYMMDD')        BETWEEN EFFECTIVE_START_DATE AND EFFECTIVE_END_DATE THEN COMMON_INSTRUMENT_KEY END)
FROM DIMENSION.COMMON_INSTRUMENT;
